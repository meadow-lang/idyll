reversed : (v : Vector Int) -> Vector Int
reversed v = case St.length (St.fromVec v) of
  (bn, a) -> let [n] = bn in let [v] = St.toVec (St.forRange 0 (n / 2) (\(i : [] Int) (m : StArray Int) -> St.swap m i (n - 1 - i)) a) in v

fibStep : (i : [] Int) -> (m : StArray Int) -> StArray Int
fibStep i m = case St.get m (i - 1) of
  (b1, m2) -> case St.get m2 (i - 2) of
    (b2, m3) -> let [x] = b1 in let [y] = b2 in St.set m3 i (x + y)

  let start = St.newArray n 0 in
  let seeded = if n > 1 then St.set start 1 1 else start in
  let [v] = St.toVec (St.forRange 2 n fibStep seeded) in v
